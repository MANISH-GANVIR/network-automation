#
# Command:
#
#    ansible-playbook playbooks/site.yml
#
# What it does:
# - Runs vlan_creation.yml
# - Runs ios_upgrade.yml
# - Runs nxos_upgrade.yml
# - All in sequence
#
# Output:
# - Creates 6 VLANs on all devices
# - Validates IOS images
# - Validates NXOS images
# - Displays results
#
#
# METHOD 2: RUN ONLY VLAN CREATION
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#
# Command:
#
#    ansible-playbook playbooks/vlan_creation.yml
#
# What it does:
# - Creates VLAN 10, 20, 30, 40, 50, 100
# - On both NEXUS and IOS devices
# - Verifies VLAN creation
# - Displays results
#
# Time: ~2-3 minutes (Dry Run)
#
#
# METHOD 3: RUN ONLY IOS UPGRADE
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#
# Command:
#
#    ansible-playbook playbooks/ios_upgrade.yml
#
# What it does:
# - Targets: ios01, ios02
# - Pre-checks: version, flash space, config backup
# - Image validation: server details, checksum
# - Configures boot variable
# - Dry run mode (no actual reboot)
#
# Time: ~3-5 minutes (Dry Run)
#
#
# METHOD 4: RUN ONLY NXOS UPGRADE
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#
# Command:
#
#    ansible-playbook playbooks/nxos_upgrade.yml
#
# What it does:
# - Targets: nexus01, nexus02 (vPC pair)
# - Pre-checks: version, vPC health, interfaces, flash
# - Image validation: server details, checksum
# - Rolling upgrade strategy
# - Upgrade secondary first (nexus02)
# - Wait for vPC recovery
# - Upgrade primary second (nexus01)
# - Final verification
#
# Time: ~5-7 minutes (Dry Run)
#
#
# METHOD 5: DRY RUN MODE (CHECK MODE - SAFE)
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Command:
#

# - Shows what WOULD happen

# - Perfect for interviews!

# - Actual output displayed
# - No configuration pushed

# ============================================================================
# UNDERSTANDING DRY RUN MODE vs REAL EXECUTION
# ============================================================================
#
# DRY RUN MODE (--check flag or dry_run_enabled: true)
# ─────────────────────────────────────────────────────
#
# What happens:
# - Tasks preview mode
# - Configuration NOT pushed to devices
# - Output shows what WOULD happen
# - Safe to run multiple times
# - Perfect for testing
#
# When to use:
# - Before production deployments
# - During testing and validation
# - In interviews (demonstrate without real devices)
# - To verify playbook logic
#
# Example:
#
#    ansible-playbook playbooks/site.yml --check
#
#
# REAL EXECUTION MODE (dry_run_enabled: false)
# ──────────────────────────────────────────────
#
# What happens:
# - Configuration pushed to devices
# - VLANs actually created
# - Devices actually rebooted
# - Real changes to network
# - CANNOT undo without manual intervention
#
# When to use:
# - Production deployments
# - After thorough testing
# - Only if you have device access
# - With proper change management
#
# Example:
#
#    # First, edit var files:
#    dry_run_enabled: false
#    skip_reboot: false
#
#    # Then run:
#    ansible-playbook playbooks/site.yml
#
# ============================================================================
# TROUBLESHOOTING
# ============================================================================
#
# ISSUE: "Error: Could not connect to device"
# ──────────────────────────────────────────
# Cause: Device IP not reachable or credentials wrong
# Fix:
#   1. Verify device IP in inventory
#   2. Check ansible_user and ansible_password
#   3. Ping device manually: ping 10.10.10.10
#   4. Test SSH: ssh admin@10.10.10.10
#
#
# ISSUE: "Module not found: cisco.nxos"
# ──────────────────────────────────────
# Cause: Collection not installed
# Fix:
#   ansible-galaxy collection install cisco.nxos cisco.ios
#
#
# ISSUE: "FAILED - SSH Error"
# ───────────────────────────
# Cause: SSH connection issues
# Fix:
#   1. Check device IP and port 22
#   2. Verify username/password
#   3. Check ansible.cfg host_key_checking setting
#   4. Test SSH manually
#
#
# ISSUE: "Task failed: image not found"
# ──────────────────────────────────────
# Cause: Image file doesn't exist on device
# Fix:
#   1. Verify image_server IP
#   2. Ensure image file copied to bootflash
#   3. Check MD5 checksum
#   4. Use "dir bootflash:" to verify
#
